// usage: node eidos-t17-check.mjs <transitions.md> <repoRoot>
import { readFileSync, existsSync, readdirSync } from "node:fs";
import { join } from "node:path";
const [file, root] = process.argv.slice(2);
const lines = readFileSync(file, "utf8").split("\n");
const groups = readdirSync(join(root, "skills")).filter(g => existsSync(join(root, "skills", g, "techniques")));
function card(id) {
  for (const g of groups) {
    const f = join(root, "skills", g, "techniques", id, "card.md");
    if (existsSync(f)) {
      const fm = (readFileSync(f, "utf8").match(/^---\n([\s\S]*?)\n---/) || [])[1] || "";
      return { path: f, kind: (fm.match(/^kind:\s*(\S+)/m) || [])[1] };
    }
  }
  return null;
}
// id column: every ascii word token; other cells: every hyphenated kebab token
function tableIds(re) {
  const s = lines.findIndex(l => re.test(l));
  if (s < 0) return null;
  const ids = new Set();
  let header = true;
  for (let i = s + 1; i < lines.length && !/^## /.test(lines[i]); i++) {
    const l = lines[i];
    if (!l.startsWith("|")) continue;
    if (/^\|\s*-/.test(l)) continue;
    if (header) { header = false; continue; }
    const cells = l.split("|").slice(1, -1);
    cells.forEach((c, k) => {
      const toks = c.match(/[a-z][a-z0-9]*(?:-[a-z0-9]+)*/g) || [];
      for (const t of toks) if (k === 1 || t.includes("-")) ids.add(t);
    });
  }
  return [...ids];
}
const A = tableIds(/^## 표 A/), B = tableIds(/^## 표 B/);
const viol = [], pend = [], ok = [];
if (!A || !A.length) viol.push("Table A missing or empty");
if (!B || !B.length) viol.push("Table B missing or empty");
function check(id, table, want) {
  if (table === "A" && (id === "camera" || id === "cut")) { ok.push(id + " (built-in)"); return; }
  const c = card(id);
  if (!c) { pend.push("PENDING " + id + " (no card yet) [Table " + table + "]"); return; }
  if (!c.kind) { pend.push("PENDING kind field (todo 6): " + id + " [Table " + table + "]"); return; }
  if (c.kind !== want) viol.push(id + " is kind " + c.kind + ", Table " + table + " needs kind " + want);
  else ok.push(id + " kind " + c.kind);
}
(A || []).forEach(id => check(id, "A", "transition"));
(B || []).forEach(id => check(id, "B", "technique"));
console.log("Table A ids (" + (A||[]).length + "): " + (A||[]).join(", "));
console.log("Table B ids (" + (B||[]).length + "): " + (B||[]).join(", "));
ok.forEach(x => console.log("OK " + x));
pend.forEach(x => console.log(x));
viol.forEach(x => console.log("VIOLATION " + x));
console.log("summary: ok=" + ok.length + " pending=" + pend.length + " violations=" + viol.length);
process.exit(viol.length ? 1 : pend.length ? 2 : 0);
